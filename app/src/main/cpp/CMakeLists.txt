cmake_minimum_required(VERSION 3.22.1)
project(localvoice_text LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
include(FetchContent)
FetchContent_Declare(llama URL https://github.com/ggml-org/llama.cpp/archive/a7a98e0fffed794396b3fbad4dcdbbc184963645.tar.gz)
FetchContent_MakeAvailable(llama)
add_library(voice_text SHARED text_jni.cpp)
target_link_libraries(voice_text PRIVATE llama android log)
target_link_options(voice_text PRIVATE "-Wl,-z,max-page-size=16384")
